# Pinned: 2.35.0 (and likely every version above 2.28.0) percent-encodes the
# Firestore "(default)" database ID in the gRPC routing header, so every
# Firestore call fails with `InvalidArgument: 400 Invalid database id
# %28default%29`. The encoding happens in google/api_core/gapic_v1/
# routing_header.py, whose urlencode() passes safe="/" — leaving "(" and ")"
# to be escaped. Not fixable in application code: explicit projectId, explicit
# database_id, and bypassing firebase_admin's wrapper to build
# google.cloud.firestore.Client directly were all verified no-ops.
# Do not bump without re-testing this specific failure mode.
google-api-core==2.28.0
google-genai==2.8.0
